dingo-cache main-fe4384a0b [local]

Usage:
  dingo-cache [OPTIONS] --id <uuid> --listen_ip <ip>

Examples:
  $ dingo-cache --id=85a4b352-... --listen_ip=10.0.0.2
  $ dingo-cache --conf cache.conf --daemonize

RUNTIME OPTIONS:
  --buffer_pool_mb=uint64                                   buffer pool per shard in MiB (default: 256)
  --cpuset=string                                           cores the shards run on, e.g. 0-31
  --idle_poll_us=uint32                                     microseconds a reactor spins on an empty queue before it sleeps; ignored under --poll_mode, which never sleeps (default: 200)
  --iodepth=uint32                                          submission queue entries per shard's ring (default: 1024)
  --offload_cpu_min_bytes=uint32                            copies below this stay on the shard (default: 32768)
  --offload_cpu_spin_us=uint32                              cpu workers spin this long for more work before parking; matches the reactor's idle_poll_us so a QD1 stream never pays a condvar wakeup per completion (default: 200)
  --offload_queue_capacity=uint32                           queued offload jobs beyond which submits answer busy (default: 4096)
  --offload_threads=uint32                                  worker threads for offloaded blocking work (default: 32)
  --pin_cpu=bool                                            pin each shard to its core (default: false)
  --poll_mode=bool                                          busy-poll the reactors (default: false)
  --shards=uint32                                           reactor shards, 0 means one per core (default: 8)
  --task_quota_us=uint32                                    microseconds a task may run before the reactor preempts it (default: 500)
